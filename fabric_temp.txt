import React, { useEffect, useRef, useState } from 'react';
import { useParams, useNavigate } from 'react-router-dom';
import * as fabric from 'fabric';
import { useYjsStore } from '../useYjsStore';
import { MousePointer2, Square, Circle, Hand, Pencil, Type, Trash2, Plus, Minus, ArrowUp, ArrowDown } from 'lucide-react';
import TopBar from '../components/TopBar';
import Sidebar from '../components/Sidebar';
import { CardComponent } from '../shapes/CardShapeUtil';
import { FolderComponent } from '../shapes/FolderShapeUtil';
import { ListComponent } from '../shapes/ListShapeUtil';
import { BoardComponent } from '../shapes/BoardShapeUtil';
import { FileComponent } from '../shapes/FileShapeUtil';
import StylePanel from '../components/StylePanel';
import BottomToolbar from '../components/BottomToolbar';
import FileViewerModal from '../FileViewerModal';
import FolderViewerModal from '../FolderViewerModal';
import BoardViewerModal from '../BoardViewerModal';
import ChartEditorModal from '../ChartEditorModal';
import ThemeSettingsModal from '../ThemeSettingsModal';
import CallManager from '../components/CallManager';
import GroupChat from '../components/GroupChat';
import TemplatesModal from '../components/TemplatesModal';
import ChoiceFileModal from '../components/ChoiceFileModal';
import UnsortedPanel from '../components/UnsortedPanel';
import HostControlPanel from '../components/HostControlPanel';

export default function FabricBoard() {
  const { id } = useParams();
  const navigate = useNavigate();
  const canvasRef = useRef(null);
  const containerRef = useRef(null);
  const htmlOverlayRef = useRef(null); // Ref for the overlay
  const fabricCanvas = useRef(null);

  // Sync state
  const storeWithStatus = useYjsStore({ 
    roomId: id || 'global-moodboard', 
    hostUrl: import.meta.env.VITE_YJS_URL || `${window.location.protocol === 'https:' ? 'wss:' : 'ws:'}//${window.location.host}/yjs`,
    hostId: null 
  });
  const [status, setStatus] = useState('connecting');
  const [activeTool, setActiveTool] = useState('select'); 
  const [activeColor, setActiveColor] = useState('yellow');
  const [activeSize, setActiveSize] = useState('M');
  const [overlayShapes, setOverlayShapes] = useState([]); // State for React overlay components 
  const [isSidebarOpen, setIsSidebarOpen] = useState(true);
  const [isChatOpen, setIsChatOpen] = useState(false);
  const [isCallActive, setIsCallActive] = useState(false);
  const [isCallHidden, setIsCallHidden] = useState(false);
  const [isMicMuted, setIsMicMuted] = useState(false);
  const [isVideoOff, setIsVideoOff] = useState(false);

  // Missing Modals and States
  const [isUploading, setIsUploading] = useState(false);
  const [uploadMessage, setUploadMessage] = useState('');
  const [showTemplatesModal, setShowTemplatesModal] = useState(false);
  const [showChoiceFileModal, setShowChoiceFileModal] = useState(false);
  const [showThemeSettings, setShowThemeSettings] = useState(false);
  const [activePreviewFile, setActivePreviewFile] = useState(null);
  const [activeBoard, setActiveBoard] = useState(null);
  const [activeIdeFolder, setActiveIdeFolder] = useState(null);
  const [activeChartEditor, setActiveChartEditor] = useState(null);
  const [activeFolderModal, setActiveFolderModal] = useState(null);
  const [roomInfo, setRoomInfo] = useState(null);
  const [isUnsortedOpen, setIsUnsortedOpen] = useState(false);
  const [unsortedItems, setUnsortedItems] = useState([]);
  
  // Host Controls
  const [showHostControls, setShowHostControls] = useState(false);
  const [hostAsk, setHostAsk] = useState(null);
  const [hostNotice, setHostNotice] = useState(null);
  const [hostLocked, setHostLocked] = useState(false);
  const [perms, setPerms] = useState({ share: true, files: true, mic: true, copyPaste: true });
  const localUserId = localStorage.getItem('userId') || (() => {
    const newId = 'user-' + Math.random().toString(36).substring(2, 9);
    localStorage.setItem('userId', newId);
    return newId;
  })();

  const yjsObjectsMap = useRef(new Map());
  const isRemoteUpdate = useRef(false);

  useEffect(() => {
    setStatus(storeWithStatus.status);
  }, [storeWithStatus.status]);

  // Modal Event Listeners
  useEffect(() => {
    const handleOpenFolderModal = (e) => setActiveFolderModal(e.detail);
    const handleOpenFileViewer = (e) => setActivePreviewFile(e.detail.file);
    const handleOpenBoardModal = (e) => setActiveBoard(e.detail);
    const handleOpenChartEditor = (e) => setActiveChartEditor(e.detail);
    
    window.addEventListener('open-folder-modal', handleOpenFolderModal);
    window.addEventListener('open-file-viewer', handleOpenFileViewer);
    window.addEventListener('open-board-modal', handleOpenBoardModal);
    window.addEventListener('open-chart-editor', handleOpenChartEditor);
    
    return () => {
      window.removeEventListener('open-folder-modal', handleOpenFolderModal);
      window.removeEventListener('open-file-viewer', handleOpenFileViewer);
      window.removeEventListener('open-board-modal', handleOpenBoardModal);
      window.removeEventListener('open-chart-editor', handleOpenChartEditor);
    };
  }, []);

  // 1. Initialize Fabric Canvas
  useEffect(() => {
    if (!canvasRef.current || !containerRef.current) return;

    const canvas = new fabric.Canvas(canvasRef.current, {
      width: containerRef.current.clientWidth,
      height: containerRef.current.clientHeight,
      backgroundColor: '#f8fafc', 
      selection: true,
      preserveObjectStacking: true, 
    });
    
    fabricCanvas.current = canvas;

    const handleResize = () => {
      canvas.setWidth(containerRef.current.clientWidth);
      canvas.setHeight(containerRef.current.clientHeight);
      canvas.renderAll();
    };
    window.addEventListener('resize', handleResize);

    return () => {
      window.removeEventListener('resize', handleResize);
      canvas.dispose();
      fabricCanvas.current = null;
    };
  }, []);

  // 2. Setup Infinite Canvas Panning and Zooming
  useEffect(() => {
    const canvas = fabricCanvas.current;
    if (!canvas) return;

    let isDragging = false;
    let lastPosX = 0;
    let lastPosY = 0;

    const syncOverlay = () => {
      if (htmlOverlayRef.current && canvas) {
        const vpt = canvas.viewportTransform;
        htmlOverlayRef.current.style.transform = `matrix(${vpt[0]}, ${vpt[1]}, ${vpt[2]}, ${vpt[3]}, ${vpt[4]}, ${vpt[5]})`;
      }
    };

    const handleMouseWheel = (opt) => {
      const delta = opt.e.deltaY;
      let zoom = canvas.getZoom();
      zoom *= 0.999 ** delta;
      if (zoom > 20) zoom = 20;
      if (zoom < 0.05) zoom = 0.05;
      canvas.zoomToPoint({ x: opt.e.offsetX, y: opt.e.offsetY }, zoom);
      syncOverlay();
      opt.e.preventDefault();
      opt.e.stopPropagation();
    };

    const handleMouseDown = (opt) => {
      if (activeTool === 'pan' || (opt.e.altKey)) {
        isDragging = true;
        canvas.selection = false;
        lastPosX = opt.e.clientX;
        lastPosY = opt.e.clientY;
      } else if (activeTool === 'text') {
        // Create text at click position
        const text = new fabric.IText('Text', {
          left: opt.pointer.x,
          top: opt.pointer.y,
          fontFamily: 'Inter',
          fill: '#000',
          fontSize: 32,
        });
        canvas.add(text);
        canvas.setActiveObject(text);
        text.enterEditing();
        text.selectAll();
        setActiveTool('select');
      }
    };

    const handleMouseMove = (opt) => {
      if (isDragging) {
        const e = opt.e;
        const vpt = canvas.viewportTransform;
        vpt[4] += e.clientX - lastPosX;
        vpt[5] += e.clientY - lastPosY;
        canvas.requestRenderAll();
        syncOverlay();
        lastPosX = e.clientX;
        lastPosY = e.clientY;
      }
    };

    const handleMouseUp = () => {
      isDragging = false;
      if (activeTool !== 'pan') {
         canvas.selection = true;
      }
      canvas.setViewportTransform(canvas.viewportTransform);
      syncOverlay();
    };

    canvas.on('mouse:wheel', handleMouseWheel);
    canvas.on('mouse:down', handleMouseDown);
    canvas.on('mouse:move', handleMouseMove);
    canvas.on('mouse:up', handleMouseUp);

    canvas.isDrawingMode = (activeTool === 'draw');
    if (activeTool === 'draw') {
      canvas.freeDrawingBrush.color = '#000';
      canvas.freeDrawingBrush.width = 4;
    }

    return () => {
      canvas.off('mouse:wheel', handleMouseWheel);
      canvas.off('mouse:down', handleMouseDown);
      canvas.off('mouse:move', handleMouseMove);
      canvas.off('mouse:up', handleMouseUp);
    };
  }, [activeTool]);

  // 3. Setup Synchronization with Yjs
  useEffect(() => {
    const canvas = fabricCanvas.current;
    const provider = storeWithStatus.provider;
    if (!canvas || !provider) return;

    const doc = provider.doc;
    const yShapes = doc.getMap('fabric-shapes');

    const syncObjectFromYjs = (id, objectData) => {
      const existingObj = yjsObjectsMap.current.get(id);
      
      if (existingObj) {
        isRemoteUpdate.current = true;
        existingObj.set(objectData);
        existingObj.setCoords();
        canvas.requestRenderAll();
        isRemoteUpdate.current = false;
      } else {
        fabric.util.enlivenObjects([objectData], (objects) => {
          const obj = objects[0];
          if (!obj) return;
          obj.id = id; 
          
          isRemoteUpdate.current = true;
          canvas.add(obj);
          yjsObjectsMap.current.set(id, obj);
          isRemoteUpdate.current = false;
          canvas.requestRenderAll();
        });
      }
    };

    const updateOverlayShapes = () => {
      const arr = [];
      yShapes.forEach((val, key) => {
        // Render any supported shape in our React overlay
        if (['milanote-folder', 'milanote-card', 'milanote-list', 'milanote-board', 'milanote-file'].includes(val.type)) {
          arr.push({ id: key, ...val });
        }
      });
      setOverlayShapes(arr);
    };

    const observeYjs = (event) => {
      let needsOverlayUpdate = false;
      event.changes.keys.forEach((change, key) => {
        if (change.action === 'add' || change.action === 'update') {
          const objectData = yShapes.get(key);
          if (objectData) {
            if (['milanote-folder', 'milanote-card', 'milanote-list', 'milanote-board', 'milanote-file'].includes(objectData.type)) {
               needsOverlayUpdate = true;
            } else {
               syncObjectFromYjs(key, objectData);
            }
          }
        } else if (change.action === 'delete') {
          needsOverlayUpdate = true;
          const objToRemove = yjsObjectsMap.current.get(key);
          if (objToRemove) {
            isRemoteUpdate.current = true;
            canvas.remove(objToRemove);
            yjsObjectsMap.current.delete(key);
            isRemoteUpdate.current = false;
          }
        }
      });
      if (needsOverlayUpdate) updateOverlayShapes();
    };

    yShapes.observe(observeYjs);

    yShapes.forEach((objectData, key) => {
      if (objectData.type === 'milanote-folder' || objectData.type === 'milanote-file') {
        // Handled by updateOverlayShapes
      } else {
        syncObjectFromYjs(key, objectData);
      }
    });
    updateOverlayShapes();

    const handleObjectModified = (e) => {
      if (isRemoteUpdate.current) return;
      
      const target = e.target;
      if (target.type === 'activeSelection') {
        target.forEachObject((obj) => {
           pushToYjs(obj);
        });
      } else {
        pushToYjs(target);
      }
    };

    const pushToYjs = (obj) => {
      if (!obj.id) {
         obj.id = `shape_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
         yjsObjectsMap.current.set(obj.id, obj);
      }
      yShapes.set(obj.id, obj.toJSON(['id'])); 
    };

    canvas.on('object:modified', handleObjectModified);
    canvas.on('path:created', (e) => {
      if (isRemoteUpdate.current) return;
      pushToYjs(e.path);
    });
    canvas.on('object:added', (e) => {
      // Don't auto-push activeSelection or paths (handled by path:created)
      if (isRemoteUpdate.current || e.target.type === 'activeSelection' || e.target.type === 'path') return;
      pushToYjs(e.target);
    });
    
    // Add delete function
    const handleDelete = (e) => {
      if (e && e.key !== 'Delete' && e.key !== 'Backspace') return;
      if (e && e.target.tagName === 'INPUT' || e && e.target.tagName === 'TEXTAREA') return;
      
      const activeObjects = canvas.getActiveObjects();
      if (activeObjects.length > 0) {
        activeObjects.forEach(obj => {
          if (obj.id) {
            yShapes.delete(obj.id); // Triggers observeYjs which removes it from canvas
          }
        });
        canvas.discardActiveObject();
        canvas.requestRenderAll();
      }
    };
    
    window.addEventListener('keydown', handleDelete);
    // Expose delete for UI button
    window._fabricDeleteSelected = handleDelete;

    return () => {
      yShapes.unobserve(observeYjs);
      canvas.off('object:modified', handleObjectModified);
      canvas.off('path:created');
      canvas.off('object:added');
      window.removeEventListener('keydown', handleDelete);
      delete window._fabricDeleteSelected;
    };
  }, [storeWithStatus.provider]);

  const addRect = () => {
    const canvas = fabricCanvas.current;
    if (!canvas) return;
    const center = canvas.getVpCenter();
    const rect = new fabric.Rect({
      left: center.x,
      top: center.y,
      fill: 'var(--accent-blue)',
      width: 100,
      height: 100,
      originX: 'center',
      originY: 'center',
      rx: 8,
      ry: 8,
      stroke: '#000',
      strokeWidth: 4
    });
    canvas.add(rect);
    setActiveTool('select');
  };

  const addShape = (type) => {
    const canvas = fabricCanvas.current;
    if (!canvas || !storeWithStatus.provider) return;
    const center = canvas.getVpCenter();
    const doc = storeWithStatus.provider.doc;
    const yShapes = doc.getMap('fabric-shapes');
    
    const id = `shape_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
    const shapeObj = {
       id,
       type,
       left: center.x - 100,
       top: center.y - 100,
       width: 250,
       height: 200,
       props: {}
    };

    if (type === 'milanote-folder') {
       shapeObj.width = 160;
       shapeObj.height = 120;
       shapeObj.props = { name: 'New Folder', files: [], roomId: 'room-' + Math.random().toString(36).substring(2, 9) };
    } else if (type === 'milanote-card') {
       shapeObj.width = 250;
       shapeObj.height = 200;
       shapeObj.props = { title: '', body: '', color: 'yellow' };
    } else if (type === 'milanote-list') {
       shapeObj.width = 300;
       shapeObj.height = 200;
       shapeObj.props = { title: 'To-Do List', items: [{ id: '1', text: 'New task', checked: false }], color: activeColor };
    } else if (type === 'milanote-file') {
       shapeObj.width = 160;
       shapeObj.height = 120;
       shapeObj.props = { fileId: '', name: 'Unknown File', url: '', isCloudFile: false, content: '' };
    }

    yShapes.set(id, shapeObj);
    setActiveTool('select');
  };

  // Function to update a folder's position inside Yjs when dragged in the overlay
  const handleOverlayDrag = (e, shape) => {
     // A complete implementation would listen to mouse events on the overlay div,
     // convert them to canvas coordinates, and update Yjs. 
     // For now, this is a placeholder to show that the overlay components are fully interactive React nodes!
     console.log("Dragging overlay shape", shape);
  };
  const addCircle = () => {
    const canvas = fabricCanvas.current;
    if (!canvas) return;
    const center = canvas.getVpCenter();
    const circle = new fabric.Circle({
      left: center.x,
      top: center.y,
      fill: 'var(--accent-pink)',
      radius: 50,
      originX: 'center',
      originY: 'center',
      stroke: '#000',
      strokeWidth: 4
    });
    canvas.add(circle);
    setActiveTool('select');
  };

  return (
    <div style={{ width: '100vw', height: '100vh', display: 'flex', flexDirection: 'column', background: '#f8fafc', overflow: 'hidden' }}>
      <TopBar 
        boardId={id} 
        boardName={`Fabric Engine [EXPERIMENTAL]`} 
        status={status}
        onToggleSidebar={() => setIsSidebarOpen(!isSidebarOpen)}
        onToggleUnsorted={() => setIsUnsortedOpen(!isUnsortedOpen)}
        onOpenHostControls={() => setShowHostControls(true)}
      />
      
      <div style={{ display: 'flex', flex: 1, position: 'relative' }}>
          <Sidebar 
            isOpen={isSidebarOpen}
            onAddShape={addShape}
            onShowTemplates={() => setShowTemplatesModal(true)}
            onShowThemeSettings={() => setShowThemeSettings(true)}
            isChatOpen={isChatOpen} setIsChatOpen={setIsChatOpen}
            isCallActive={isCallActive} setIsCallActive={setIsCallActive}
            isCallHidden={isCallHidden} setIsCallHidden={setIsCallHidden}
            isMicMuted={isMicMuted} setIsMicMuted={setIsMicMuted}
            isVideoOff={isVideoOff} setIsVideoOff={setIsVideoOff}
          />
          
          {/* Unsorted Panel */}
          <UnsortedPanel 
            isOpen={isUnsortedOpen}
            onClose={() => setIsUnsortedOpen(false)}
            unsortedItems={unsortedItems}
            onDragStart={(e, item) => {
              e.dataTransfer.setData('application/unsorted-item', JSON.stringify(item));
              e.dataTransfer.effectAllowed = 'move';
            }}
            onDrop={(items) => {}}
            onDragOver={(e) => { e.preventDefault(); }}
          />

          {/* Canvas Container */}
        <div ref={containerRef} style={{ flex: 1, position: 'relative', overflow: 'hidden' }}>
        <canvas ref={canvasRef} />
        
        {/* HTML Overlay Engine */}
        <div 
          style={{ 
            position: 'absolute', 
            top: 0, left: 0, right: 0, bottom: 0, 
            pointerEvents: 'none', 
            overflow: 'hidden' 
          }}
        >
          <div 
            ref={htmlOverlayRef}
            style={{ 
              position: 'absolute', 
              top: 0, left: 0, 
              width: '100%', height: '100%', 
              transformOrigin: '0 0'
            }}
          >
            {/* React Components will be rendered here perfectly synced to the canvas zoom/pan */}
            {overlayShapes.map(shape => (
               <div 
                 key={shape.id}
                 style={{
                    position: 'absolute',
                    left: shape.left,
                    top: shape.top,
                    width: shape.width,
                    height: shape.height,
                    pointerEvents: 'all', // Re-enable pointer events for the React component
                 }}
                 onMouseDown={(e) => handleOverlayDrag(e, shape)}
               >
                 {shape.type === 'milanote-folder' && (
                    <FolderComponent 
                      shape={shape} 
                      updateShape={({ id, props }) => {
                        const yShapes = storeWithStatus.provider.doc.getMap('fabric-shapes');
                        const s = yShapes.get(id);
                        if (s) yShapes.set(id, { ...s, props: { ...s.props, ...props } });
                      }}
                    />
                 )}
                 {shape.type === 'milanote-card' && (
                    <CardComponent 
                      shape={shape} 
                      updateShape={({ id, props }) => {
                        const yShapes = storeWithStatus.provider.doc.getMap('fabric-shapes');
                        const s = yShapes.get(id);
                        if (s) yShapes.set(id, { ...s, props: { ...s.props, ...props } });
                      }}
                    />
                 )}
                 {shape.type === 'milanote-list' && (
                    <ListComponent 
                      shape={shape} 
                      updateShape={({ id, props }) => {
                        const yShapes = storeWithStatus.provider.doc.getMap('fabric-shapes');
                        const s = yShapes.get(id);
                        if (s) yShapes.set(id, { ...s, props: { ...s.props, ...props } });
                      }}
                    />
                 )}
                 {shape.type === 'milanote-board' && (
                    <BoardComponent 
                      shape={shape} 
                      updateShape={({ id, props }) => {
                        const yShapes = storeWithStatus.provider.doc.getMap('fabric-shapes');
                        const s = yShapes.get(id);
                        if (s) yShapes.set(id, { ...s, props: { ...s.props, ...props } });
                      }}
                    />
                 )}
                 {shape.type === 'milanote-file' && (
                    <FileComponent 
                      shape={shape} 
                      updateShape={({ id, props }) => {
                        const yShapes = storeWithStatus.provider.doc.getMap('fabric-shapes');
                        const s = yShapes.get(id);
                        if (s) yShapes.set(id, { ...s, props: { ...s.props, ...props } });
                      }}
                    />
                 )}
                 {!['milanote-folder', 'milanote-card', 'milanote-list', 'milanote-board', 'milanote-file'].includes(shape.type) && (
                   <div style={{ background: '#fff', border: '4px solid #000', width: '100%', height: '100%', display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
                     {shape.type}
                   </div>
                 )}
               </div>
            ))}
          </div>
        </div>
                {/* Context Menu / Quick Actions for Selected Shape */}
        {activeTool === 'select' && (
          <div style={{
            position: 'absolute', bottom: '80px', left: '50%', transform: 'translateX(-50%)',
            display: 'flex', gap: '8px', background: 'var(--color-panel)',
            padding: '8px 12px', borderRadius: '12px', border: '3px solid #000',
            boxShadow: '4px 4px 0 #000', pointerEvents: 'all', zIndex: 1000
          }}>
            <button className="neo-btn" style={{ padding: '6px' }} onClick={() => {
              const active = fabricCanvas.current?.getActiveObjects();
              if (active && active.length) {
                active.forEach(obj => fabricCanvas.current.bringObjectForward(obj));
                fabricCanvas.current.renderAll();
              }
            }} title="Bring Forward">
              <ArrowUp size={16} />
            </button>
            <button className="neo-btn" style={{ padding: '6px' }} onClick={() => {
              const active = fabricCanvas.current?.getActiveObjects();
              if (active && active.length) {
                active.forEach(obj => fabricCanvas.current.sendObjectBackwards(obj));
                fabricCanvas.current.renderAll();
              }
            }} title="Send Backward">
              <ArrowDown size={16} />
            </button>
            <button className="neo-btn" style={{ padding: '6px' }} onClick={() => {
              const active = fabricCanvas.current?.getActiveObjects();
              if (active && active.length) {
                alert('Duplicate coming soon!');
              }
            }} title="Duplicate">
              <Square size={16} />
            </button>
            <button className="neo-btn" style={{ padding: '6px', color: 'var(--accent-red)' }} onClick={() => {
              const active = fabricCanvas.current?.getActiveObjects();
              if (active && active.length) {
                active.forEach(obj => {
                  fabricCanvas.current.remove(obj);
                  if (obj.id) yjsObjectsMap.current.delete(obj.id);
                });
                fabricCanvas.current.discardActiveObject();
                fabricCanvas.current.renderAll();
              }
            }} title="Delete">
              <Trash2 size={16} />
            </button>
          </div>
        )}

        {/* Zoom UI Widget */}
          <div style={{
            position: 'absolute', bottom: '24px', left: '24px', zIndex: 1000,
            display: 'flex', alignItems: 'center', background: 'var(--color-panel)',
            borderRadius: '8px', border: '3px solid #000', boxShadow: '4px 4px 0 #000',
            padding: '4px', gap: '4px', pointerEvents: 'all'
          }}>
            <button className="neo-btn" style={{ width: '32px', height: '32px', padding: 0 }} onClick={() => {
              if (fabricCanvas.current) {
                let zoom = fabricCanvas.current.getZoom() * 0.8;
                fabricCanvas.current.zoomToPoint({ x: fabricCanvas.current.width/2, y: fabricCanvas.current.height/2 }, zoom);
              }
            }} title="Zoom Out">
              <Minus size={16} />
            </button>
            <div style={{ width: '48px', textAlign: 'center', fontSize: '13px', fontWeight: 'bold' }}>
              100%
            </div>
            <button className="neo-btn" style={{ width: '32px', height: '32px', padding: 0 }} onClick={() => {
              if (fabricCanvas.current) {
                let zoom = fabricCanvas.current.getZoom() * 1.2;
                fabricCanvas.current.zoomToPoint({ x: fabricCanvas.current.width/2, y: fabricCanvas.current.height/2 }, zoom);
              }
            }} title="Zoom In">
              <Plus size={16} />
            </button>
          </div>
        
        {/* Bottom Toolbar & Panels */}
        <StylePanel activeColor={activeColor} setActiveColor={setActiveColor} activeSize={activeSize} setActiveSize={setActiveSize} />
        <BottomToolbar activeTool={activeTool} setActiveTool={setActiveTool} addShape={addShape} />

        <div style={{
          position: 'absolute',
          bottom: '20px',
          right: '20px',
          display: 'flex',
          gap: '8px'
        }}>
          <button 
            className="neo-btn" 
            onClick={() => {navigate(`/tldraw-board/${id}`)}}
            style={{ background: 'var(--accent-purple)', padding: '12px', color: '#fff' }}
            title="Switch back to old Tldraw engine"
          >
            Revert to old engine
          </button>
        </div>
      </div>

      {/* Modals & Portals */}
      {showThemeSettings && (
        <ThemeSettingsModal 
          onClose={() => setShowThemeSettings(false)}
        />
      )}
      {activePreviewFile && (
        <FileViewerModal 
          file={activePreviewFile} 
          onClose={() => setActivePreviewFile(null)} 
        />
      )}
      {activeFolderModal && (
        <FolderViewerModal 
          folderShape={activeFolderModal.shape} 
          editor={null} 
          onClose={() => setActiveFolderModal(null)} 
        />
      )}
      {activeBoard && (
        <BoardViewerModal 
          boardShape={activeBoard.shape} 
          editor={null} 
          onClose={() => setActiveBoard(null)} 
        />
      )}
      {activeChartEditor && (
        <ChartEditorModal 
          shape={activeChartEditor.shape} 
          editor={null} 
          onClose={() => setActiveChartEditor(null)} 
        />
      )}
      {showHostControls && (
        <HostControlPanel
          onClose={() => setShowHostControls(false)}
          localUserId={localUserId}
          roomId={id}
        />
      )}
      {isCallActive && (
        <CallManager
          roomId={id}
          isCallHidden={isCallHidden}
          isMicMuted={isMicMuted}
          isVideoOff={isVideoOff}
          localUserId={localUserId}
          userName={"You"}
        />
      )}
      {isChatOpen && (
        <GroupChat
          roomId={id}
          localUserId={localUserId}
          userName={"You"}
          onClose={() => setIsChatOpen(false)}
        />
      )}
      {showTemplatesModal && (
        <TemplatesModal 
          onClose={() => setShowTemplatesModal(false)}
          onSelectTemplate={(tid) => alert('Templates coming soon to Fabric!')}
        />
      )}
      {showChoiceFileModal && (
        <ChoiceFileModal 
          onClose={() => setShowChoiceFileModal(false)}
        />
      )}
    </div>
  );
}
